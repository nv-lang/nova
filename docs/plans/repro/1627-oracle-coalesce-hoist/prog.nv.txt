fn mark(s str, v int) -> int {
    println("eval ${s}")
    v
}

fn maybe(v int) -> Option[int] {
    if v < 0 { return None }
    Some(v)
}

fn main() {
    ro a = mark("f", 1) + (maybe(mark("g", 2)) ?? 0)
    println(a)
    ro b = mark("l", 0) > 1 && (maybe(mark("never", 5)) ?? 0) > 3
    println(b)
    ro c = mark("l2", 2) > 1 || (maybe(mark("never2", 5)) ?? 0) > 3
    println(c)
}
