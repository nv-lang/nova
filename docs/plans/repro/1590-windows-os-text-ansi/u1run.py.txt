import subprocess, os, sys
exe = sys.argv[1]
word = "Евг"          # three Cyrillic letters, UTF-8: d0 95 d0 b2 d0 b3
env = dict(os.environ, U1_PROBE=word)
r = subprocess.run([exe, word], env=env, capture_output=True)
print(r.stdout.decode("utf-8", "replace").strip())
print("expected hex for both: d095d0b2d0b3 (UTF-8 of the three letters)")
