module bc2

fn main() Io -> () {
    ro s = "12.5e3"
    match s.to_f64() {
        Ok(v) => println("${v}")
        Err(_) => println("err")
    }
    match "42".to_int() {
        Ok(v) => println("${v}")
        Err(_) => println("err")
    }
    match "-7".to_i8() {
        Ok(v) => println("${v}")
        Err(_) => println("err")
    }
    match "ff".to_int(radix: 16) {
        Ok(v) => println("${v}")
        Err(_) => println("err")
    }
    match "true".to_bool() {
        Ok(v) => println("${v}")
        Err(_) => println("err")
    }
}
