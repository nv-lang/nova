module standalone.repro1020_mono_self_return

type Slot[T] {
    mut items []T
    mut room int
}

fn Slot[T].new() -> Slot[T] => { items: []T.new(), room: 0 }

// The shape under suspicion: a method that returns the receiver (`-> @`) on a generic type,
// so a concrete instantiation has to be monomorphised.
fn Slot[T] mut @with_room(n int) -> @ {
    @room = n
    @
}

fn Slot[T] mut @put(v T) -> @ {
    @items.push(v)
    @
}

type Cell { mut n int }

fn main() Io -> () {
    mut a = Slot[int].new()
    a.with_room(3).put(1)
    print("${a.room} ")
    mut b = Slot[Cell].new()
    b.with_room(7).put({ n: 1 })
    print("${b.room}")
}
