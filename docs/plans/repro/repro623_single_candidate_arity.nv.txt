// Проба №623: единственный кандидат с НЕВЕРНЫМ ЧИСЛОМ аргументов.
//
// Два сайта, названные отчётом окна p617 и оставленные им нетронутыми:
//   (а) голое имя свободной функции   — types/mod.rs ~15163
//   (б) статический `Type.method(...)` — types/mod.rs ~15352
//
// До правки быстрый путь `Some([single]) => single` возвращал кандидата
// БЕЗУСЛОВНО: чекер говорил `ok`, а кодоген выпускал C-вызов, не сходящийся с
// прототипом по числу параметров. Ровно тот класс, что закрыт у третьего брата
// (№618), и ровно та беда, которую там подтвердили саботаж-пробой.
//
// Ожидание ПОСЛЕ правки: оба вызова отвергаются на чекере.
module probe623

type Box {
    v int
}

export fn Box.make(a int, b int) -> Box => { v: a + b }

fn only_one(a int, b int) -> int => a + b

fn main() {
    // (а) свободная функция: объявлена с двумя параметрами, зовём с одним.
    println("free=${only_one(1)}")

    // (б) статический метод: объявлен с двумя, зовём с одним.
    ro bx = Box.make(1)
    println("static=${bx.v}")
}
