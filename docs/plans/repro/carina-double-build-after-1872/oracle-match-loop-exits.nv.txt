// Reduced diagnostic probe for #1876; the full original is preserved separately.
// Expected result: 43. The oracle accepts check but emits n as nova_unit.
module standalone.p1875_oracle_match_loop_exits

fn loop_exits() -> int {
    mut out = 0
    for i in 0..3 {
        ro n = match i {
            0 => { continue }
            1 => { if true { 43 } else { break } }
            _ => { break }
        }
        out += n
    }
    out
}

fn main() Io -> () {
    println(loop_exits())
    assert(loop_exits() == 43)
}
