module standalone.probe_sumwrap_roundtrip

type Tok { text str }

type Nd enum
    | Leaf(Tok)
    | Branch { children []Nd }

fn leaf_text(n Nd) -> str {
    match n {
        Leaf(t) => t.text
        Branch { .. } => "not a leaf"
    }
}

fn main() Io -> () {
    ro tok = Tok { text: "ZERO" }
    ro node = Nd.Leaf(tok)
    match node {
        Leaf(t) => {
            print("A:")
            print(t.text)           // печатает "ZERO" -- t.text напрямую цел
            print("B:")
            print(leaf_text(t))     // ожидание "ZERO", факт: "" (пусто)
        }
        Branch { .. } => print("branch")
    }
}

// Наблюдённый вывод (побайтово, через xxd, exit=0, stderr пуст):
//   A:ZEROB:
// -- то есть после "B:" ничего вообще, ни байта, ни новой строки.
