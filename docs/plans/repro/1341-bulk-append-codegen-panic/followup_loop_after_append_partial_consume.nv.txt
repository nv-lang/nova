// Seen fixing #1341 (not caused by it; the consume pass does not read append's body):
// `v.append(tail)` / `v.append(mid)` with `consume` bindings, then a `for i in 0..v.len()`
// loop reading `v.view(i, ..)` -- nova check: [D133-not-consumed] `tail` / `mid`
// "consumed only on part of the execution paths". Without the loop (or with the same
// append on a `Vec[Res].new()` receiver) the program checks clean.
// EXPECT_STDOUT made=9 closed=9 sum=45
//
// Registry 221.1 #1341: the consuming `append` / `splice` of `Vec[T consume]`
// move `other`'s elements with ONE bulk copy (`RawMem.copy_n`, D216
// amendment 2026-09-25: a bulk move is outside linearity and the module keeps
// it correct) instead of a per-element `for consume` loop. Every `Res` must
// change owner exactly once and be closed exactly once: made = closed, and
// the ids 1..9 sum to 45. `order` shows where each moved element landed.
//
// A copying `append` / `splice` on an ordinary `Vec[int]` passed a `mut`
// binding must leave that binding intact (rule 3 of the D84 mode axis,
// 10-overloading.md: the consuming candidate is excluded for a ro/mut
// binding) -- `other=2` and `kept=2`. Before the fix the consuming form was
// chosen there too and the bulk move emptied the caller's live vector.

type Res consume {
    id int
}

fn Res consume @close() -> int => @id

fn make(from int, to int, mut made []int) -> Vec[Res] {
    consume v = Vec[Res].new()
    for i in from..to {
        v.push(Res { id: i })
        made.push(i)
    }
    v
}

fn main() Io -> () {
    mut made []int = []
    consume v Vec[Res] = make(1, 3, made)
    consume tail Vec[Res] = make(3, 7, made)
    v.append(tail)
    consume mid Vec[Res] = make(7, 10, made)
    v.append(mid)
    mut order = ""
    for i in 0..v.len() {
        ro id = v.view(i, |r| r.id)
        order = if i == 0 { "${id}" } else { "${order},${id}" }
    }
    mut closed = 0
    mut sum = 0
    for consume r in v {
        sum += r.close()
        closed += 1
    }

    print("made=${made.len()} closed=${closed} sum=${sum} order=${order}")
}
