// КОНТРОЛЬ 2 к №1185 (добавлен по требованию интегратора): `Option` НЕ над
// указателем.
//
// Зачем: он держит будущий фикс честным. Правка, снёсшая обёртку у ВСЕХ `Option`,
// выглядела бы успехом на первых двух пробах; здесь структура с тегом — норма, и
// она обязана остаться. То есть приёмка проверяет обе стороны: у указателя
// обёртка исчезает, у не-указателя сохраняется.

fn takeint(h Option[int]) -> int => match h { Some(v) => v, None => 0 }

fn main() Io -> () {
    print("ok ${takeint(None)}")
}
