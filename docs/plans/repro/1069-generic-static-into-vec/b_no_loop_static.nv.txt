// b: цикла НЕТ, статик по тип-параметру ЕСТЬ. Ожидание и факт: собирается.
// Отличается от `d` ровно отсутствием `for` — это и есть проба в обе стороны.

type Row { ro vals []str }

type FromRow protocol {
    .from_row(r Row) -> Self
}

type Account value { ro email str }

fn Account.from_row(r Row) -> Account => { email: r.vals[0] }

fn[T FromRow] boxed(r Row) -> Vec[T] {
    mut out = Vec[T].new()
    out.push(T.from_row(r))     // статик по тип-параметру + push в Vec[T]
    return out
}

fn mk() -> Row => { vals: ["a@b.c"] }

fn main() Io -> () {
    ro v = boxed[Account](mk())
    print("${v[0].email}")
}
