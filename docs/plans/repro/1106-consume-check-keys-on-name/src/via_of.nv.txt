// `ro` binding of a consume-required value built via `of`.
// D180 says this needs `consume`, so the ERROR is the correct behaviour and a PASS
// here means the check was skipped.

module probe_consume_name.via_of

import probe_consume_name.secret.{Secret, build_secret}

export fn take() -> str {
    ro t = Secret.of("x")
    "${t:?}"
}
