// Замер цены лени для плана 200 §23 (окно p200-23, 2026-09-15).
// Bench-инфраструктура закрыта чужим дефектом (bench/micro/strings.nv красный
// на main), поэтому замер — прямой программой через Monotonic.
//
// Вопрос ОДИН: платит ли ленивость там, где потребителю нужны ПЕРВЫЕ ДВА слова
// длинной строки. Жадная форма (то, что вынуждала подпись `-> ro []str`)
// обязана построить весь Vec; ленивая останавливается после второго слова.

import std.time.duration
import std.collections.vec_iter

fn long_line(words int) -> str {
    consume sb = StringBuilder.new()
    mut i = 0
    while i < words {
        sb.append("word ")
        i = i + 1
    }
    sb.into_str()
}

fn main() Io Time -> () {
    ro haystack = long_line(2000)
    ro iters = 50

    // A — ленивая форма: два слова и стоп.
    ro a0 = Monotonic.now()
    mut i = 0
    mut sink_a = 0
    while i < iters {
        ro two = haystack.split(" ").take(2).collect()
        sink_a = sink_a + two.len()
        i = i + 1
    }
    ro a = Monotonic.now().elapsed_since(a0)

    // B — жадная форма: весь массив, затем два слова из него.
    ro b0 = Monotonic.now()
    mut j = 0
    mut sink_b = 0
    while j < iters {
        ro all = haystack.split(" ").collect()
        sink_b = sink_b + all.len()
        j = j + 1
    }
    ro b = Monotonic.now().elapsed_since(b0)

    // C — контроль: когда нужны ВСЕ слова, лень не должна стоить дороже.
    ro c0 = Monotonic.now()
    mut k = 0
    mut sink_c = 0
    while k < iters {
        mut it = haystack.split(" ")
        mut n = 0
        for _w in it { n = n + 1 }
        sink_c = sink_c + n
        k = k + 1
    }
    ro c = Monotonic.now().elapsed_since(c0)

    print("iters=${iters} words=2000")
    print("A lazy take(2)      : ${a.micros()} us  (sink ${sink_a})")
    print("B eager collect-all : ${b.micros()} us  (sink ${sink_b})")
    print("C lazy for-in all   : ${c.micros()} us  (sink ${sink_c})")
}
