// КОНТРОЛЬ к пробе №699: столкновение, которое ОБЯЗАНО быть замечено.
//
// Основная проба показывает, что имя `Node` из ТЕСТОВОГО peer-файла std больше
// не сталкивается с типом потребителя. Сам по себе этот результат ничего не
// стоит: так же выглядела бы проба, в которой столкновение физически не могло
// возникнуть. Поэтому здесь имя берётся у ЭКСПОРТИРОВАННОГО типа того же
// модуля — `SerError`, — и отказ на нём доказывает, что путь столкновений жив,
// а молчание на `Node` есть ответ, а не слепота.
module p699probe

import std.encoding.serde.{json_encode, SerError}

type SerError { ro k int }

fn main() Io -> () {
    ro e SerError = { k: 1 }
    println("k=${e.k}")
}
