// Потребитель `encoding.serde`, объявляющий СВОЙ тип с именем `Node` —
// тем же, что объявлен в тестовом peer-файле модуля std
// (`std/src/encoding/serde/tagging_test.nv`, `type Node enum`).
//
// Улика к строке реестра 221.1 №699. Ничего специального здесь нет: это ровно
// то, что напишет обычный пользователь std, у которого в программе есть тип с
// распространённым именем.
module p699probe

import std.encoding.serde.{json_encode, SerError}

type Node { ro n int }

fn main() Io -> () {
    // `= { … }` без повтора имени типа: компилятор отвергает избыточный
    // префикс, когда тип уже назван в аннотации. Проверено им же.
    ro x Node = { n: 7 }
    // Serde ВЫЗЫВАЕТСЯ по-настоящему, а не только импортируется. Первая
    // редакция пробы импортировала `json_encode` и не звала его — компилятор
    // сказал «unused import», то есть модуль мог и не попасть в единицу
    // компиляции, и «столкновения нет» читалось бы как ответ, хотя предмет в
    // пробу не входил. Пустая мишень выглядит как чистая.
    ro r = json_encode("probe")
    if Ok(txt) = r { println("serde ok: ${txt}, node n=${x.n}") }
}
