fn use_h(h Effect[Random]) -> int => 1

fn mk() -> Effect[Random] {
    ro h = effect Random {
        u64() -> u64 => 4
        bytes(n int) -> []u8 => "ab".bytes()
    }
    h
}

fn mk2(k int) -> Effect[Random] {
    ro h = effect Random {
        u64() -> u64 => 4
        bytes(n int) -> []u8 => "ab".bytes()
    }
    if k == 0 {
        return h
    }
    h
}

fn g() -> int {
    ro h = effect Random {
        u64() -> u64 => 4
        bytes(n int) -> []u8 => "ab".bytes()
    }
    use_h(h)
}

fn main() {
    println(g())
}
