// #1327 carrier: `clear` of a `[]Res` drops the elements unconsumed -- nova check PASS.
type Res consume {
    id int
}

fn Res consume @close() -> () { () }

fn mk(i int) -> Res => { id: i }

fn drain(consume v []Res) -> () {
    for consume r in v {
        r.close()
    }
}

fn main() -> () {
    mut v = []Res.new()
    v.push(mk(1))
    v.push(mk(2))
    v.clear()
    drain(v)
}
