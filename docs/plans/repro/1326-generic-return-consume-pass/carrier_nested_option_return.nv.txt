// nova check: W_CONSUME_KEYWORD_UNNECESSARY + D133-not-consumed on `o` (type `Option`, inner unknown).
type Res consume {
    id int
}

fn Res consume @close() -> () {
}

fn wrap[T consume](consume x T) -> Option[T] => Some(x)

fn main() -> () {
    consume o = wrap(Res { id: 1 })
    match o {
        Some(consume r) => { r.close() }
        None => { }
    }
}
