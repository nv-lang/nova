// ПРОБА e — КОНТРОЛЬ, и она обязана быть ЗЕЛЁНОЙ.
//
// Зачем. Пробы a–d показали ось «элемент коллекции»: []int проходит,
// []<плоская запись> падает. Но между ними лежит ещё одна форма, и от неё
// зависит вся граница плана 286: коллекция СУММЫ с полезной нагрузкой.
// `SqlValue` — сумма, `ColumnDef` — запись, и если сумма проходит, то держит
// не «строка таблицы», а именно СХЕМА, что сужает дефект вдвое.
//
// Эта проба — не поиск красноты, а КОНТРОЛЬ: если она упадёт, вывод «ось —
// элемент» неверен, и виновата коллекция как таковая.
//
// Запуск:
//   nova-cli/target/release/nova build <этот файл, переименованный в .nv>
// Судить ТОЛЬКО по `build`: `nova check` на всём семействе зелёный, в том
// числе там, где кодоген падает (это и есть суть строки №1070).

type Val enum
    | Null
    | Int(i64)
    | Text(str)

fn mk() -> []Val => [Val.Int(7), Val.Text("a@b.c"), Val.Null]

type Db effect {
    bind(args []Val) -> int
    back()           -> []Val
}

fn main() Io -> () {
    with Db = effect Db {
        bind(args) -> int   => args.len()
        back()     -> []Val => mk()
    } {
        print("in=${Db.bind(mk())}")
        ro out = Db.back()
        match out[1] {
            Val.Text(s) => print("out=${s}")
            _           => print("out=other")
        }
    }
}
