// ПРОБА g — КАНДИДАТНАЯ ФОРМА ГРАНИЦЫ ПЛАНА 286.
//
// Зачем. Если e зелёная, а c/f красные, то ось дефекта №1070 — не «коллекция»
// и не «строка таблицы», а ЭЛЕМЕНТ: запись падает, сумма проходит. Тогда у
// плана 286 остаётся ровно один законный способ провести схему через границу —
// закодировать колонку СУММОЙ, а не записью.
//
// Что здесь проверяется. Колонка несёт три вещи (имя, тип, nullable). Сумма
// несёт их вариантом с полезной нагрузкой: `Col.Text(name, nullable)`. Это не
// косметика — это отказ от записи в пользу суммы В САМОЙ ГРАНИЦЕ, а фасад над
// эффектом волен собрать из неё привычную запись, где vtable уже ни при чём.
//
// ПОЧЕМУ НЕ ОБХОД ПО ИНДЕКСУ. Напрашивающийся обход — вернуть непрозрачную
// ручку и вычитывать колонки по одной (`@col_count` + `@col_at(i)`). Он
// ЗАПРЕЩЁН D456, раздел «Чего на границе эффекта быть не должно», п. 3,
// дословно: «Три операции обязаны договориться между собой: мок отдаст пять
// ключей и четыре значения — и не покраснеет НИЧЕГО. Правильно — одна
// операция, возвращающая коллекцию». Поэтому проба и ищет форму, где операция
// ОДНА и возвращает коллекцию.
//
// Ожидание — ЗЕЛЕНО. Красный здесь дороже красного в f: он означает, что
// законной формы границы у 286 нет вовсе, и план упирается в кодоген.
//
// Запуск:
//   nova-cli/target/release/nova build <этот файл, переименованный в .nv>
// Судить ТОЛЬКО по `build`: `nova check` зелёный на всём семействе.

type Col enum
    | Text(str, bool)
    | Int(str, bool)

type Val enum
    | Null
    | Text(str)
    | Int(i64)

fn schema() -> []Col => [Col.Int("id", false), Col.Text("email", true)]
fn row()    -> []Val => [Val.Int(7), Val.Text("a@b.c")]

type Db effect {
    describe() -> []Col
    next()     -> Option[[]Val]
}

fn name_of(c Col) -> str => match c {
    Col.Text(n, _) => n
    Col.Int(n, _)  => n
}

fn main() Io -> () {
    with Db = effect Db {
        describe() -> []Col         => schema()
        next()     -> Option[[]Val] => Some(row())
    } {
        ro cols = Db.describe()
        print("cols=${cols.len()} first=${name_of(cols[0])}")
        match Db.next() {
            Some(vs) => match vs[1] {
                Val.Text(s) => print("val=${s}")
                _           => print("val=other")
            }
            None => print("none")
        }
    }
}
