// #1330 carrier: `for x in v { take(x) }` -- consume through a view (D157 forbids it), then drained -- nova check PASS.
// Run: nova check <file>. Element 0 is closed twice when the carrier passes.
type Res consume {
    id int
}

fn Res consume @close() -> () {
}

fn take(consume r Res) -> () {
    r.close()
}

fn drain_all(consume v Vec[Res]) -> () {
    for consume x in v {
        x.close()
    }
}

fn main() -> () {
    consume v = Vec[Res].new()
    v.push(Res { id: 1 })
    for x in v {
        take(x)
    }
    drain_all(v)
}
