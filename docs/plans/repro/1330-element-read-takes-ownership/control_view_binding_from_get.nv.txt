// #1330 control: `ro r = v.get(0)` is a view -- PASS, and it is correct.
// Run: nova check <file>. Element 0 is closed twice when the carrier passes.
type Res consume {
    id int
}

fn Res consume @close() -> () {
}

fn take(consume r Res) -> () {
    r.close()
}

fn drain_all(consume v Vec[Res]) -> () {
    for consume x in v {
        x.close()
    }
}

fn main() -> () {
    consume v = Vec[Res].new()
    v.push(Res { id: 1 })
    ro r = v.get(0)
    drain_all(v)
}
