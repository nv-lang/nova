// Реестр 221.1 №762: `match` по `str` с образцом-КОНСТАНТОЙ.
// Ожидание по строке: check проходит, сборка падает тремя ошибками C —
// скрутинизируемое `str` эмитится как СУММА, имя константы как имя ВАРИАНТА.
module a_const_pattern

const MUT_W = "mut"

fn classify(w str) -> int => match w {
    MUT_W => 1
    _ => 0
}

fn main() Io -> () {
    ro a = classify("mut")
    ro b = classify("ro")
    println("mut=${a} other=${b}")
}
