// Registry 1870 (red): a LOCAL declared in one operation body of a handler literal, then a later
// operation whose PARAMETER has the same name and another type, forwarded to a function.
module repro1870.leak_test

import std.os.{Signal}

type SigFx effect {
    fn waitish(c int) -> (int, int)
    fn num(s Signal) -> int
}

fn n_of(s Signal) -> int {
    match s {
        Interrupt => 2
        Terminate => 15
        Kill => 9
        Hangup => 1
        Other(n) => 1000 + n
    }
}

fn h() -> Effect[SigFx] {
    effect SigFx {
        waitish(c int) -> (int, int) {
            mut s = 0
            s = c + 1
            (c, s)
        }
        num(s Signal) -> int => n_of(s)
    }
}

fn via(s Signal) SigFx -> int => SigFx.num(s)

test "a local of an earlier operation body does not retype a later parameter" {
    with SigFx = h() {
        println("kill=${via(Signal.Kill)} other=${via(Signal.Other(42))}")
        assert(via(Signal.Kill) == 9)
        assert(via(Signal.Other(42)) == 1042)
    }
}
