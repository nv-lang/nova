// Registry 921, form 4 of 4: the OTHER direction -- too FEW arguments. `@get(i int)`
// called as `b.get()`. Passes `nova check`. Filed separately from the "too many"
// forms because a fix that counts one way and not the other would close the row by
// carrier, and because the two produce different C failures.
module p921_too_few

type Box priv { n int }
fn Box.new() -> Box => { n: 0 }
fn Box @get(i int) -> int => @n + i

fn main() Io -> () {
    ro b = Box.new()
    println(b.get())
}
