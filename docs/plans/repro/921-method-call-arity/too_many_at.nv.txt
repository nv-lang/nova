// Registry 921, form 1 of 4: `@take(1, 2)` -- the `@` self-call form, one argument
// too many. `nova check` says ok; `nova build` fails in C with
// `too many arguments to function call`.
module p921_too_many_at

type Box priv { n int }
fn Box.new() -> Box => { n: 0 }
fn Box mut @take(n int) -> () { @n = n }
fn Box mut @drive() -> () { @take(1, 2) }

fn main() Io -> () {
    mut b = Box.new()
    b.drive()
    println(b.n)
}
