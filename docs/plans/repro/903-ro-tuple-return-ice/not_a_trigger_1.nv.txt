// Registry 903, ОТРИЦАТЕЛЬНОЕ свидетельство #1: эта форма СОБИРАЕТСЯ.
// Простой именованный кортеж, возврат `ro`-параметра, разбор результата в main.
// Записано, потому что «минимальная проба не воспроизвела» — это факт о ФОРМЕ,
// а не о находке: чинящий иначе начнёт ровно с неё и потеряет час.
module not_a_trigger_1

type Pair(a int, b int)

fn f(p Pair) -> ro (Pair, int) {
    return (p, 0)
}

fn main() Io -> () {
    ro q = Pair(1, 2)
    ro (r, n) = f(q)
    println("${r.a} ${n}")
}
