// Ось: БЕЗ newtype вовсе — обычная запись, реализующая протокол. Контроль
// для утверждения «дело в newtype».
module c_plain_record

type Ordinal protocol {
    @ordinal() -> int
    .from_ordinal(i int) -> Self
}
type Cell { v int }
fn Cell @ordinal() -> int => @v
fn Cell.from_ordinal(i int) -> Cell => { v: i }
fn mk[I Ordinal](n int) -> I => I.from_ordinal(n)
fn main() Io -> () {
    ro r = mk[Cell](41)
    println(r.ordinal() + 1)
}
