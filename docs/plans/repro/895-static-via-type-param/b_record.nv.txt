// подстановка ПРИМИТИВНОГО представления.
module b_record

type Ordinal protocol {
    @ordinal() -> int
    .from_ordinal(i int) -> Self
}
type Cell { v int }
type Wrap Cell
fn Wrap @ordinal() -> int => (@ as Cell).v
fn Wrap.from_ordinal(i int) -> Wrap => Cell { v: i } as Wrap
fn mk[I Ordinal](n int) -> I => I.from_ordinal(n)
fn main() Io -> () {
    ro r = mk[Wrap](41)
    println(r.ordinal() + 1)
}
