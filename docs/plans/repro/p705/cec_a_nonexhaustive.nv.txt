// №705 — приёмочная проба: коллизия голых имён СЛЕПИТ проверку исчерпаемости.
//
// Здесь объявлен `CecKind` с ТРЕМЯ вариантами. В соседнем файле того же CU —
// `CecKind` с ДВУМЯ, и оба попадают в карту типов чекера, ключуемую голым
// именем: побеждает тот, кто зарегистрировался позже.
//
// Проверка исчерпаемости (№703) несёт самопроверку разрешения: если арм назвал
// вариант, которого в поднятой сумме нет, значит поднята НЕ ТА сумма, и
// проверка снимается ЦЕЛИКОМ. Это честный обход — но его цена в том, что при
// коллизии исчерпаемость не проверяется вовсе, и непокрытый вариант проходит
// молча (реестр №705, «цена обхода, названная честно»).
//
// Приёмка окна W6: разрешение идёт по файлу МЕСТА ИСПОЛЬЗОВАНИЯ, поднимается
// СВОЙ `CecKind`, самопроверка не срабатывает, и `neg`-половина этой пары
// начинает честно краснеть.
module crossmod_exhaustive_collision.cec_a

type CecKind enum Alpha | Beta | Gamma

// ЗДЕСЬ НЕТ ВАРИАНТА `Gamma` — ожидается E_MATCH_NON_EXHAUSTIVE.
// Сегодня отказа НЕТ: коллизия слепит проверку (реестр №705).
export fn describe_a(k CecKind) -> int {
    match k {
        Alpha => 1
        Beta => 2
    }
}
