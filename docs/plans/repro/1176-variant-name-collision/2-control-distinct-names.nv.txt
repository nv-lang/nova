// позиционным) или конструирование обоих в одной программе. Контроль
// отличается от пробы РОВНО одной буквой в имени второго варианта.

type Rv enum Pay { vec int } | Pos(int)

fn main() Io -> () {
    ro a = Rv.Pay { vec: 1 }
    ro b = Rv.Pos(2)
    match a {
        Pay { vec } => print("rec ${vec}")
    }
    match b {
        Pos(n) => print("pos ${n}")
    }
}
