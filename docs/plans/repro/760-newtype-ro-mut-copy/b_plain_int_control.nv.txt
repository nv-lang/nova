// КОНТРОЛЬ. Тот же цикл, та же mut-копия из ro-параметра, но носитель --
// ПЛОСКИЙ `int`, без newtype. Проходит.
//
// Без этого файла проба `a_both_forms` доказывала бы лишь «что-то отвергнуто»,
// а не «отвергнуто ИЗ-ЗА обёртки»: форма цикла, mut-копия и ro-параметр здесь
// те же самые, отличается ровно одно -- newtype снят.
//
// Копия с голым `.nv` в скрэтч, затем `nova check`:
//   -> PASS: 1  FAIL: 0

fn walk(at int, stop int) -> int {
    mut cur = at
    mut n = 0
    while cur != stop {
        cur = cur + 1
        n += 1
    }
    n
}

fn main() Io -> () => print("${walk(0, 3)}")
