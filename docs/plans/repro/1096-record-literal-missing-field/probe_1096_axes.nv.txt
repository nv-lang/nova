// №1096, две оставшиеся оси — обе к компилятору, а не к рассуждению.
//
// (A) ПОЛЕ С УМОЛЧАНИЕМ пропущено — это ЗАКОННО, и правило звучит «обязательно каждое
//     поле БЕЗ умолчания», а не «все поля». Если реализовать буквально «все», покраснеет
//     здоровая форма, и страж, краснеющий на здоровом, долго не живёт.
//
// (B) ЛИШНЕЕ поле — `02-types.md:1956` объявляет ошибкой так же, как и пропущенное.
//     Диагностики в списке `E_RECORD_*` я не вижу; проба скажет, одна это дверь или две.
//
// Обе оси в ОДНОМ файле намеренно: если (B) окажется ошибкой компиляции, весь файл не
// соберётся, и это само по себе ответ — но тогда (A) останется непроверенной, поэтому
// (B) вынесена в комментарий и проверяется отдельным прогоном.

type Complexish(re f64 = 0.0, im f64 = 0.0)

type Pointish(x int = 7, y int)

fn main() Io -> () {
    // (A1) оба поля с умолчанием, литерал пуст
    ro a = Complexish()
    print("all-default omitted: re=")
    print(a.re)
    print(" im=")
    print(a.im)
    print("\n")

    // (A2) одно поле с умолчанием пропущено, другое названо
    // D102: поле БЕЗ умолчания у скобочной формы передаётся ПОЗИЦИОННО — это
    // подсказал сам компилятор, и это ДРУГАЯ форма, чем фигурный литерал №1096.
    ro b = Pointish(5)
    print("default field omitted: x=")
    print(b.x)
    print(" y=")
    print(b.y)
    print("\n")
}
