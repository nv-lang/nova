// #1332 carrier (axis 2, dispatch): the same with a must-consume `Res` temporary -- form=1, the copying form ran on a linear value.
// EXPECT_STDOUT x
module standalone.q3_res_temporary

type Res consume {
    id int
}

fn Res consume @close() -> int => @id

type Bag[T consume] {
    n int
}

fn Bag[T consume] consume @done() -> () {
}

// copying form: ordinary T only
fn Bag[T] mut @put(v T) -> int => 1

// consuming form: any T
fn Bag[T consume] mut @put(consume v T) -> int {
    keep(v)
    2
}

fn keep[T consume](consume x T) -> () {
    sink(x)
}

fn sink[T consume](consume x T) -> () {
    keep(x)
}

fn main() Io -> () {
    consume b Bag[Res] = { n: 0 }
    ro k = b.put(Res { id: 1 })
    b.done()
    print("form=${k}")
}
