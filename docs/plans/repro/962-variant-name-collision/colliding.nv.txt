// Registry TBD (new carrier of #136): two sums in one CU share the variant name `Invalid`
// and `AboveMax`; the constructor is written QUALIFIED and still lowers to the wrong sum.
// Expected stdout: "e -> Invalid at 3", "big -> AboveMax". Measured 2026-09-05 (oracle):
// "e -> other", "big -> Invalid at 0".
module colliding

type Flag enum Ok2 | Invalid              // a unit `Invalid` declared FIRST in the CU
type Range2 enum AboveMax | BelowMin     // `AboveMax` at tag 0 here

type ParseErr enum Empty | Invalid { at int } | AboveMax | BelowMin

fn parse(s str) -> Result[int, ParseErr] {
    if s == "" { return Err(ParseErr.Empty) }
    if s == "big" { return Err(ParseErr.AboveMax) }
    Err(ParseErr.Invalid { at: 3 })
}

fn main() Io -> () {
    match parse("e") {
        Err(ParseErr.Invalid { at }) => println("e -> Invalid at ${at}")
        Err(_) => println("e -> other")
        Ok(_) => println("e -> ok?")
    }
    match parse("big") {
        Err(ParseErr.AboveMax) => println("big -> AboveMax")
        Err(ParseErr.Invalid { at }) => println("big -> Invalid at ${at}")
        _ => println("big -> other")
    }
}
