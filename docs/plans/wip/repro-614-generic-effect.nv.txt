module scratch_ge.probe_ge2

type Store[T] effect {
    put(v T) -> ()
    get() -> T
}

fn main() {
    ro ints = effect Store[int] {
        put(_v T) -> () => ()
        get() -> T => 7
    }
    ro strs = effect Store[str] {
        put(_v T) -> () => ()
        get() -> T => "seven"
    }

    with Store[int] = ints {
        print(Store[int].get())
    }
    with Store[str] = strs {
        print(Store[str].get())
    }
}
