// Реестр 221.1 №613. Проба интегратора 2026-08-12.
//
// Значение метода (`Type.@method`, D-форма из spec/decisions/03-syntax.md)
// НЕ удерживало метод живым: обход достижимости ходит по именам-вызовам и не
// считает взятие значения использованием. Метод вычищался, а переходник
// значения на него ссылался — `lld-link: error: undefined symbol:
// Nova_Purse_method_add`. Это тот же корень, что и №576: у методов и функций
// в компиляторе разные места (план 273).
//
// Фикстура ЗОВЁТ методы ТОЛЬКО через значения — прямых вызовов здесь нет
// намеренно: именно прямой вызов и прятал дефект (с ним всё линкуется).

module standalone.method_value_keeps_method_alive

type Purse {
    balance int
}

fn Purse @add(n int) -> int => @balance + n

fn Purse mut @bump(n int) -> int {
    @balance = @balance + n
    return @balance
}

fn main() {
    mut a = Purse { balance: 43 }

    ro h = Purse.@add
    print(h(a, 10))
    print("|")

    // Аннотация типа БЕЗ маркера получателя: `@` помечает получателя в
    // ОБЪЯВЛЕНИИ, в тип он не входит.
    ro h2 fn(Purse, int) -> int = Purse.@add
    print(h2(a, 5))
    print("|")

    ro g fn(mut Purse, int) -> int = Purse.@bump
    print(g(a, 1))
    print("|")
}
