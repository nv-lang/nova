type Color enum
    | Red
    | Green

fn g(o Option[Color]) -> int => match o {
    Some(Red) => 1
    Some(Green) => 2
    None => 0
}

fn mk(f bool, c Color) -> Option[Color] => if f { Some(c) } else { None }

fn main() {
    println(g(mk(true, Color.Red)))
    println(g(mk(true, Color.Green)))
    println(g(mk(false, Color.Red)))
}
