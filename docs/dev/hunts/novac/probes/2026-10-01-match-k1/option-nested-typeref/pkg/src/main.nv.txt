// Type arguments read by the one door (typeref.nv, 7bbe42250): an Option of
// an Option, matched level by level -- the inner value matched again by its
// own `match` (no nested pattern, which is #1542).
fn mk(f bool, n int) -> Option[int] => if f { Some(n) } else { None }

fn mk2(f bool, g bool) -> Option[Option[int]] => if f { Some(mk(g, 3)) } else { None }

fn d2(o Option[Option[int]]) -> int => match o {
    Some(inner) => match inner {
        Some(x) => x
        None => -1
    }
    None => -2
}

fn main() {
    println(d2(mk2(true, true)))
    println(d2(mk2(true, false)))
    println(d2(mk2(false, true)))
}
