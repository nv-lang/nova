// Result as std hands it back (the two shell-carried calls, `read` and
// `read_dir`), with the Err payload READ -- a value record (IoError), its
// fields used in the arm -- and a guard and a binder of the whole Result.
// Run from the repository root (run-probe.sh does): `AGENTS.md` exists there.
import std.fs.{read, read_dir, real_fs}

fn op_of(p str) Fs -> str => match read(p.to_path()) {
    Ok(_) => "ok"
    Err(e) => e.op
}

fn raw(p str) Fs -> int => match read(p.to_path()) {
    Ok(b) if b.len() > 10 => 1
    Ok(_) => 2
    Err(e) => if e.raw_os != 0 { 3 } else { 4 }
}

fn dir_or(p str) Fs -> str => match read_dir(p.to_path()) {
    Err(e) => "err ${e.op}"
    whole => "whole"
}

fn main() {
    with Fs = real_fs() {
        println(op_of("AGENTS.md"))
        println(op_of("no_such_file_hunt_1001.txt"))
        println(raw("AGENTS.md"))
        println(raw("no_such_file_hunt_1001.txt"))
        println(dir_or("novac"))
        println(dir_or("no_such_dir_hunt_1001"))
    }
}
