// The scrutinee of a string match and of an Option match is a CALL with a
// visible effect: it must be evaluated once, however many arms test it.
fn word(k int) -> str {
    println("word ${k}")
    if k == 1 { "one" } else { "many" }
}

fn opt(k int) -> Option[int] {
    println("opt ${k}")
    if k > 0 {
        return Some(k)
    }
    None
}

fn w(k int) -> int => match word(k) {
    "zero" => 0
    "one" => 1
    "two" => 2
    _ => 9
}

fn o(k int) -> int => match opt(k) {
    None => -1
    Some(x) if x > 5 => 50
    Some(x) => x
}

fn main() {
    println(w(1))
    println(w(3))
    println(o(0))
    println(o(7))
    println(o(2))
}
