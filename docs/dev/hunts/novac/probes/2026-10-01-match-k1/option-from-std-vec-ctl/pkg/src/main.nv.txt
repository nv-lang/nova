// Control for ../option-from-std-vec: the same instance bound first and
// matched by name -- if check refuses here, the match-scrutinee position is
// the only one where the instance passes check unjudged.
fn ff(d []f64) -> f64 {
    ro o = d.first()
    match o {
        Some(x) => x
        None => 0.0
    }
}

fn main() {
    println(ff([1.5, 2.5]))
}
