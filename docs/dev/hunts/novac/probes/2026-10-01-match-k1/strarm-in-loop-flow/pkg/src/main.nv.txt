// String and char arms inside a loop, with `continue`, `break` and `return`
// in the arm bodies: the first-match flag must be fresh on every iteration.
fn count(words []str) -> int {
    mut n = 0
    for w in words {
        match w {
            "skip" => continue
            "stop" => break
            _ => {
                n = n + 1
            }
        }
        n = n + 100
    }
    n
}

fn first_a(cs []char) -> int {
    mut i = 0
    for c in cs {
        match c {
            'a' => return i
            _ => {}
        }
        i = i + 1
    }
    -1
}

fn main() {
    println(count(["a", "skip", "b", "stop", "c"]))
    println(count(["x", "y"]))
    println(first_a(['x', 'y', 'a', 'a']))
    println(first_a(['x']))
}
