fn k(s str) -> int => match s {
    "\r\n" => 2
    "\t" => 3
    "\$x" => 4
    "%d%s" => 5
    "a\\" => 6
    "??=" => 7
    _ => 0
}

fn main() {
    println(k("\r\n"))
    println(k("\n"))
    println(k("\t"))
    println(k("\$x"))
    println(k("%d%s"))
    println(k("a\\"))
    println(k("??="))
    println(k("#"))
}
