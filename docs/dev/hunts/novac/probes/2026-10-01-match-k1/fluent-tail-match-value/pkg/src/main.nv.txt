// A `-> @` body whose TAIL is a value-yielding `match` and `if` -- D409: a
// tail that is not the receiver is discarded as a statement, then `return @`.
type Ctr {
    n int
}

fn side(x int) -> int {
    println("side ${x}")
    x
}

fn Ctr mut @m(k int) -> @ {
    @n = @n + k
    match k {
        1 => side(10)
        _ => side(20)
    }
}

fn Ctr mut @i(k int) -> @ {
    @n = @n * k
    if k > 2 { side(30) } else { side(40) }
}

fn main() {
    mut c = Ctr { n: 1 }
    println(c.m(1).n)
    println(c.i(3).n)
}
