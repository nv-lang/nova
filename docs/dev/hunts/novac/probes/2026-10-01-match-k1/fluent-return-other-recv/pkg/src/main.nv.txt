// `-> @` with an explicit `return <value of the receiver's type>` that is not
// the receiver, in a BLOCK body (D409 rule 2: returning anything but the
// receiver is an error). Both compilers accept it and return `o` -- the
// statement-body twin of ../fluent-expr-body-other-recv.
type Ctr {
    n int
}

fn Ctr mut @swap(mut o Ctr, k int) -> @ {
    @n = @n + k
    if k > 10 {
        return o
    }
}

fn main() {
    mut a = Ctr { n: 1 }
    mut b = Ctr { n: 50 }
    println(a.swap(b, 2).n)
    println(a.swap(b, 20).n)
    println(a.n)
}
