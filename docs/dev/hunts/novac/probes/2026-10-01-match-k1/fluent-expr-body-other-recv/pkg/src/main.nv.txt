// `-> @` whose expression body is a value of the receiver's TYPE that is not
// the receiver: D409 makes the receiver the only possible return (rule 2:
// "returning anything but the receiver is an error"; bullet 3: a tail that is
// not the receiver is discarded, then `return @`).
type Ctr {
    n int
}

fn Ctr mut @swap(mut o Ctr) -> @ => o

fn Ctr mut @fresh() -> @ => { n: 99 }

fn main() {
    mut a = Ctr { n: 1 }
    mut b = Ctr { n: 2 }
    println(a.swap(b).n)
    println(a.fresh().n)
    println(a.n)
}
