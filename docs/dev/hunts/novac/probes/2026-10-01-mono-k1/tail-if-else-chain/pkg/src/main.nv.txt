module hp

fn deep(a int, b int) -> int {
    if a > 0 {
        if b > 0 { 1 } else { 2 }
    } else {
        0
    }
}

fn byte_at(b []u8, k int) -> u8 {
    if k < b.len() {
        b[k]
    } else {
        0
    }
}

fn pair(a int) -> (int, int) {
    if a > 0 {
        (a, 1)
    } else {
        (0, 2)
    }
}

fn grade(n int) -> str {
    if n > 90 {
        "A"
    } else if n > 50 {
        "B"
    } else {
        "C"
    }
}

fn sign(n int) -> int {
    ro s = if n > 0 { 1 } else if n < 0 { -1 } else { 0 }
    s * 10
}

fn pick(o Option[int], c bool) -> int {
    if Some(v) = o {
        v
    } else if c {
        1
    } else {
        2
    }
}

fn main() {
    println(pick(Some(7), true))
    println(pick(None, true))
    println(pick(None, false))
    println(grade(95))
    println(grade(70))
    println(grade(10))
    println(sign(-4))
    println(sign(0))
    println(deep(1, 2))
    println(deep(1, -2))
    println(deep(-1, 2))
    ro b = "AB".bytes()
    println(byte_at(b, 1) as int)
    println(byte_at(b, 5) as int)
    ro (x, y) = pair(5)
    println(x + y)
}
