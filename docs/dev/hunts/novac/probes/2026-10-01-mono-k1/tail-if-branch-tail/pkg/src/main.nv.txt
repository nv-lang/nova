module hp

fn deep(a int, b int) -> int {
    if a > 0 {
        if b > 0 { 1 } else { 2 }
    } else {
        0
    }
}

fn byte_at(b []u8, k int) -> u8 {
    if k < b.len() {
        b[k]
    } else {
        0
    }
}

fn pair(a int) -> (int, int) {
    if a > 0 {
        (a, 1)
    } else {
        (0, 2)
    }
}

fn main() {
    println(deep(1, 2))
    println(deep(1, -2))
    println(deep(-1, 2))
    ro b = "AB".bytes()
    println(byte_at(b, 1) as int)
    println(byte_at(b, 5) as int)
    ro (x, y) = pair(5)
    println(x + y)
}
