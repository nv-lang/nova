module hp

fn span(a int) -> Option[(int, int)] {
    match a {
        0 => None
        _ => Some((a, 2))
    }
}

fn whole(a int) -> Option[(int, int)] => Some((a, 3))

fn main() {
    ro (x, y) = span(1) ?? (0, 0)
    println(x + y)
    ro (p, q) = whole(4) ?? (0, 0)
    println(p + q)
    ro (u, w) = span(0) ?? (5, 6)
    println(u + w)
}
