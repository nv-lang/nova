type AppErr enum
    | Io(str)
    | Code(int)

fn get(n int) -> Result[int, str] => if n > 0 { Ok(n) } else { Err("neg") }

fn work(n int) -> Result[int, AppErr] {
    ro v = get(n)?
    Ok(v + 1)
}

fn main() {
    match work(3) {
        Ok(v) => println("ok ${v}")
        Err(e) => match e {
            Io(s) => println("io ${s}")
            Code(c) => println("code ${c}")
        }
    }
    match work(-3) {
        Ok(v) => println("ok ${v}")
        Err(e) => match e {
            Io(s) => println("io ${s}")
            Code(c) => println("code ${c}")
        }
    }
}
