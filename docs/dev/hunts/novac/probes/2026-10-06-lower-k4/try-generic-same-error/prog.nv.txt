fn get(n int) -> Result[int, int] => if n > 0 { Ok(n) } else { Err(-n) }

fn pass[E](r Result[int, E]) -> Result[int, E] {
    ro v = r?
    Ok(v + 1)
}

fn main() {
    match pass(get(3)) {
        Ok(v) => println("ok ${v}")
        Err(e) => println("err ${e}")
    }
    match pass(get(-3)) {
        Ok(v) => println("ok ${v}")
        Err(e) => println("err ${e}")
    }
}
