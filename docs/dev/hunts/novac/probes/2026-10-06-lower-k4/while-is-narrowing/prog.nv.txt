fn main() {
    ro base = 40
    with Supervisor = effect Supervisor {
        on_child_fail(idx, err) -> Decision {
            while err is str {
                println("caught: ", err)
                break
            }
            Decision.Stop
        }
    } {
        supervised {
            spawn {
                println(base / (base - 40))
            }
        }
    }
    println("after")
}
