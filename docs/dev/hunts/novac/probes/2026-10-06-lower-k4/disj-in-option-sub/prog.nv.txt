fn inner(k int) -> Option[int] {
    if k == 1 {
        return None
    }
    Some(k)
}

fn mk(k int) -> Option[Option[int]] {
    if k == 0 {
        return None
    }
    Some(inner(k))
}

fn f(x Option[Option[int]]) -> str => match x {
    Some(None | Some(2)) => "a"
    Some(Some(_)) => "b"
    None => "c"
}

fn main() {
    println(f(mk(1)))
    println(f(mk(2)))
    println(f(mk(5)))
    println(f(mk(0)))
}
