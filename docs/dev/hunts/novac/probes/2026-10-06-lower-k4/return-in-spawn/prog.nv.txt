fn work(n int) -> Result[int, int] {
    supervised {
        spawn {
            if n < 0 {
                return Err(n)
            }
            println("in spawn ${n}")
        }
    }
    Ok(n)
}

fn main() {
    match work(3) {
        Ok(v) => println("ok ${v}")
        Err(e) => println("err ${e}")
    }
    match work(-3) {
        Ok(v) => println("ok ${v}")
        Err(e) => println("err ${e}")
    }
}
