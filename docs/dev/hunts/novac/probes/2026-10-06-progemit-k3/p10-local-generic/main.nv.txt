module p10_local_generic.main

// NOVAC_PROGRAM
// A generic DECLARED in `mid`, instantiated over `geo.Pt` inside `mid`
// itself (not re-imported from elsewhere): isolates whether the ICE needs
// the generic AND its instantiation site to live in DIFFERENT modules from
// the substituted type's declaration, or fires whenever `[]T` is
// substituted with a type imported from another module at all.

import ./geo.{Pt}
import ./mid.{wrap}

fn main() {
    mut xs = []Pt.new()
    xs.push(Pt { x: 1, y: 2 })
    xs.push(Pt { x: 3, y: 4 })
    println(wrap(xs))
}
