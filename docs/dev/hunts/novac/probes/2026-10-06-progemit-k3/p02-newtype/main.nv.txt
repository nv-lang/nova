module p02_newtype.main

// NOVAC_PROGRAM
// A newtype declared in a leaf module, used as a record field's type by the
// entry: does the field spell the newtype's representation (nova_int), the
// same way the leaf module itself would, or does it diverge by module?

import ./libn.{Row, mk_row}

export type Holder {
    r Row
}

fn main() {
    ro h = Holder { r: mk_row(7) }
    println(h.r)
}
