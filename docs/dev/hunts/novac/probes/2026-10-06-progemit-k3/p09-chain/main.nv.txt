module p09_chain.main

// NOVAC_PROGRAM
// A three-module chain: entry -> mid -> geo. `mid` instantiates geo.count
// over geo.Pt, a type it only knows through re-import; the entry builds the
// Vec[Pt] itself and ALSO instantiates count[Pt] directly. Does the instance
// made through the chain match the one made directly by the entry?

import ./geo.{Pt, count}
import ./mid.{wrap}

fn main() {
    mut xs = []Pt.new()
    xs.push(Pt { x: 1, y: 2 })
    xs.push(Pt { x: 3, y: 4 })
    println(wrap(xs))
    println(count[Pt](xs))
}
