module p04_vecfield.main

// NOVAC_PROGRAM
// Vec over another module's record, held as a FIELD of the entry's own record:
// does the field's typedef (the Vec instance of geo.Pt) match the one the
// entry's own expressions build, one copy in the program?

import ./geo.{Pt, mk}

export type Group {
    pts []Pt
}

fn main() {
    mut xs = []Pt.new()
    xs.push(mk(1, 2))
    xs.push(mk(3, 4))
    ro g = Group { pts: xs }
    println(g.pts.len())
    println(g.pts[0].x)
    println(g.pts[1].y)
}
