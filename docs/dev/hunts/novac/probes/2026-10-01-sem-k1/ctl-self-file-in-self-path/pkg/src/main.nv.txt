module hp

type Box { v int }

fn Box @get() -> int => @v

fn show(x int) -> str => "int ${x}"
fn show(x str) -> str => "str ${x}"

fn pick[T](a T, b T, first bool) -> T => if first { a } else { b }

fn main() {
    println(show(3))
    println(show("q"))
    println(pick(1, 2, false))
    println(pick("a", "b", true))
    ro b = Box { v: 9 }
    println(b.get())
}
