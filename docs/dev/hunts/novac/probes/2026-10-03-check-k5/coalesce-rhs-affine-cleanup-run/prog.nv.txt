module p

type Tx consume {
    id int
}

fn Tx consume @commit() -> () {
    println("commit ${@id}")
}

fn ok(consume t Tx) -> bool {
    t.commit()
    true
}

fn eat(consume t Tx) -> int {
    t.commit()
    0
}

fn mk(n int) -> Option[int] => if n > 0 { Some(n) } else { None }

fn Tx consume @cleanup(outcome ScopeOutcome) -> () {
    println("cleanup ${@id}")
}

fn run(n int) -> () {
    consume t = Tx { id: n }
    ro v = mk(n) ?? eat(t)
    println(v)
}

fn main() {
    run(1)
    run(0)
}
