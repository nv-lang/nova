module p

type Tx consume {
    id int
}

fn Tx consume @commit() -> () {
    println("commit ${@id}")
}

fn ok(consume t Tx) -> bool {
    t.commit()
    true
}

fn eat(consume t Tx) -> int {
    t.commit()
    0
}

fn mk(n int) -> Option[int] => if n > 0 { Some(n) } else { None }

fn run(n int) -> () {
    for i in 0..2 {
        consume t = Tx { id: i }
        for j in 0..2 {
            if j == 1 {
                break
            }
        }
        t.commit()
    }
}

fn main() {
    run(1)
}
