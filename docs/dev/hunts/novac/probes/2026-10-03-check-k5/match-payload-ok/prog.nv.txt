module p

type Tx consume {
    id int
}

fn Tx consume @commit() -> () {
    println("commit ${@id}")
}

fn ok(consume t Tx) -> bool {
    t.commit()
    true
}

fn eat(consume t Tx) -> int {
    t.commit()
    0
}

fn mk(n int) -> Option[int] => if n > 0 { Some(n) } else { None }

fn mkt(n int) -> Option[Tx] => if n > 0 { Some(Tx { id: n }) } else { None }

fn run(n int) -> () {
    match mkt(n) {
        Some(consume x) => x.commit()
        None => println("none")
    }
}

fn main() {
    run(1)
    run(0)
}
