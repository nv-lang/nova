module p

type Tx consume {
    id int
}

fn Tx consume @commit() -> () {
    println("commit ${@id}")
}

fn ok(consume t Tx) -> bool {
    t.commit()
    true
}

fn eat(consume t Tx) -> int {
    t.commit()
    0
}

fn mk(n int) -> Option[int] => if n > 0 { Some(n) } else { None }

fn run(n int) -> () {
    consume t = Tx { id: n }
    if n == 1 {
        t.commit()
    } else if n == 2 {
        t.commit()
    }
}

fn main() {
    run(1)
}
