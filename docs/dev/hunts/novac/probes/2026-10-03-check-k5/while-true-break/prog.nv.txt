module p

type Tx consume {
    id int
}

fn Tx consume @commit() -> () {
    println("commit ${@id}")
}

fn ok(consume t Tx) -> bool {
    t.commit()
    true
}

fn eat(consume t Tx) -> int {
    t.commit()
    0
}

fn mk(n int) -> Option[int] => if n > 0 { Some(n) } else { None }

fn run(n int) -> () {
    consume t = Tx { id: n }
    while true {
        if n > 0 {
            break
        }
        t.commit()
        return
    }
}

fn main() {
    run(1)
}
