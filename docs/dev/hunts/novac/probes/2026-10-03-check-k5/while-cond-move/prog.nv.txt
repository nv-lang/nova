module p

type Tx consume {
    id int
}

fn Tx consume @commit() -> () {
    println("commit ${@id}")
}

fn ok(consume t Tx) -> bool {
    t.commit()
    true
}

fn mk(n int) -> Option[int] => if n > 0 { Some(n) } else { None }

fn run(n int) -> () {
    consume t = Tx { id: n }
    mut i = 0
    while i < 2 && ok(t) {
        i += 1
    }
}

fn main() {
    run(1)
}
