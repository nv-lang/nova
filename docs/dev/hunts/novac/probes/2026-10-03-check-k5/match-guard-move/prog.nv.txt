module p

type Tx consume {
    id int
}

fn Tx consume @commit() -> () {
    println("commit ${@id}")
}

fn ok(consume t Tx) -> bool {
    t.commit()
    true
}

fn mk(n int) -> Option[int] => if n > 0 { Some(n) } else { None }

fn run(n int) -> () {
    consume t = Tx { id: n }
    match n {
        1 if ok(t) => println("one")
        _ => println("other")
    }
}

fn main() {
    run(2)
}
