// Hunt 2026-10-02 cast x K1, side probe (outside K1: a LOUD refusal).
// `bool as <number>` is legal since the D54 amendment 2026-10-01 p.1, and a
// cast of a literal is a constant initializer in Carina for every other
// legal pair (`300 as u8`, `'A' as u8`, `3.9 as int` -- all three below are
// accepted). Only the bool literal is refused as E_CONST_NOT_CONSTEXPR.
const A = 300 as u8
const C = 'A' as u8
const B = 3.9 as int
const D = true as u8

fn main() {
    println(A)
    println(C)
    println(B)
    println(D)
}
