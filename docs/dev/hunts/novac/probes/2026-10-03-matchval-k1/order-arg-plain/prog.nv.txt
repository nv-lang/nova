module prog

// D484 control: call arguments are evaluated left to right, observably --
// no `match`, no `if`: three plain calls as the three arguments.
fn say(n int) -> int {
    println(n)
    n
}

fn three(a int, b int, c int) -> int => a * 100 + b * 10 + c

type Box { n int }

fn Box @add3(a int, b int, c int) -> int => @n + a + b + c

fn main() {
    println("plain call, norm 1 2 3 then 123")
    println(three(say(1), say(2), say(3)))
    println("method call, norm 4 5 6 then 15")
    ro b = Box { n: 0 }
    println(b.add3(say(4), say(5), say(6)))
    println("two-arg, norm 7 8 then 78")
    println(three(0, say(7), say(8)))
}
