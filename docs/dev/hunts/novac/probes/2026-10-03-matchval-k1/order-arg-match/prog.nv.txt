module prog

// D484: call arguments are evaluated left to right, observably.
// The middle argument is a `match` whose scrutinee and arm have effects.
fn say(n int) -> int {
    println(n)
    n
}

fn three(a int, b int, c int) -> int => a * 100 + b * 10 + c

fn main() {
    println("case 1, norm 1 2 3 then 123")
    println(three(say(1), match say(2) {
        2 => 2
        _ => 0
    }, say(3)))
    println("case 2, norm 4 5 6 then 456")
    println(three(say(4), match 1 {
        1 => say(5)
        _ => 0
    }, say(6)))
    println("case 3, norm 7 8 9 then 789 (if for control)")
    println(three(say(7), if say(8) > 0 { 8 } else { 0 }, say(9)))
}
