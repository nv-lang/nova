module prog

// D491: "a position with a written type -- a declaration, a call argument, a
// return, a RECORD FIELD, a collection element: the value's type equals the
// position's; otherwise an error, the hint is the form with `as`". Here an
// `int` VALUE (a local, not a literal) is given to a `u8` field and to an
// `f64` field: plainly, through an `if` value and through a `match` value (the
// sweep-9 position). The norm: no program -- the oracle refuses every line.
// Each print is labelled with what a silent conversion would show.
type Pv { b u8, f f64 }

fn main() {
    ro k = 3
    ro n = 300
    ro q1 = Pv { b: n, f: 2.0 }
    println("plain int 300 into u8 field (no program by norm)")
    println(q1.b as int)
    ro q2 = Pv { b: if k > 0 { n } else { n }, f: 2.0 }
    println("if-value int 300 into u8 field (no program by norm)")
    println(q2.b as int)
    ro q3 = Pv { b: match k {
        3 => n
        _ => n
    }, f: 2.0 }
    println("match-value int 300 into u8 field (no program by norm)")
    println(q3.b as int)
    ro q4 = Pv { b: 1 as u8, f: match k {
        3 => k
        _ => k
    } }
    println("match-value int 3 into f64 field (no program by norm)")
    println(q4.f)
}
