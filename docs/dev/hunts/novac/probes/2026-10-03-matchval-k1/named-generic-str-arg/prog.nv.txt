module prog

// A `match` value as the argument of a call whose
// parameter is GENERIC (the type comes from the match), with `str` and
// record arms, and as a method argument whose receiver is itself a `match`
// value bound first. Pure arms; each print labelled with the norm.
type Tally { n int }

fn Tally @plus(k int) -> int => @n + k

fn first[T](a T, b T) -> ro T => a

fn found(k int) -> Option[int] {
    if k > 0 { Some(k) } else { None }
}

fn main() {
    ro k = 3
    println("generic T from match arms (str), norm three")
    println(first(match k {
        3 => "three"
        _ => "other"
    }, "x"))
    println("generic T from match arms (record), norm 7")
    ro t = first(match found(k) {
        Some(v) => Tally { n: v + 4 }
        None => Tally { n: 0 }
    }, Tally { n: 9 })
    println(t.n)
    println("method arg with binding arm, receiver from a match, norm 37")
    ro acc = match k {
        3 => Tally { n: 7 }
        _ => Tally { n: 0 }
    }
    println(acc.plus(match found(k) {
        Some(v) => v * 10
        None => 0
    }))
    println("str arm into println directly, norm three")
    println(match k {
        3 => "three"
        _ => "other"
    })
}
