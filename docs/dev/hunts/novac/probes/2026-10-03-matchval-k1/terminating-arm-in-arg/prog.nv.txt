module prog

// A `match` value whose arm LEAVES the function (`return`) or the loop
// (`break`/`continue`), standing as an argument, an element and the right of
// `+=`. A leaving arm produces no value: the call / store must not run on
// that path. Each print is labelled with the norm.
fn found(k int) -> Option[int] {
    if k > 0 { Some(k) } else { None }
}

fn twice(n int) -> int {
    println("twice called")
    n * 2
}

fn pick(k int) -> int {
    ro r = twice(match found(k) {
        Some(v) => v
        None => return -7
    })
    r + 1
}

fn main() {
    println("pick(5): twice called, then 11")
    println(pick(5))
    println("pick(0): -7, no call")
    println(pick(0))
    println("loop sum over 3 0 4 with continue on None: 7")
    mut sum = 0
    for k in [3, 0, 4] {
        sum += match found(k) {
            Some(v) => v
            None => continue
        }
    }
    println(sum)
    println("loop stops at 0 with break: 3")
    mut acc = 0
    for k in [3, 0, 4] {
        ro t = twice(match found(k) {
            Some(v) => v
            None => break
        })
        acc = acc + t / 2
    }
    println(acc)
}
