module prog

// A `match` on the right of `=` / `+=` / `-=` / `*=` whose LEFT is a record
// field or an array element, the arms binding (`Some(v) => v`) and ending in
// blocks. Each line is labelled with the value the norm requires.
type Purse { n int, m int }

fn found(k int) -> Option[int] {
    if k > 0 { Some(k) } else { None }
}

fn main() {
    ro k = 3
    mut a = Purse { n: 10, m: 1 }
    a.n += match found(k) {
        Some(v) => v * 100
        None => -1
    }
    println("a.n += some, norm 310")
    println(a.n)
    a.m -= match found(0) {
        Some(v) => v
        None => {
            ro z = 5
            z + 1
        }
    }
    println("a.m -= none-block, norm -5")
    println(a.m)
    mut xs = [1, 2, 3]
    xs[1] += match k {
        3 => 40
        _ => 0
    }
    println("xs[1] += 40, norm 42")
    println(xs[1])
    xs[2] *= match found(k) {
        Some(v) => v
        None => 0
    }
    println("xs[2] *= 3, norm 9")
    println(xs[2])
    xs[0] = match found(k) {
        Some(v) => v + 1000
        None => 0
    }
    println("xs[0] = 1003, norm 1003")
    println(xs[0])
    a.n = match match k {
        3 => found(7)
        _ => found(0)
    } {
        Some(v) => v
        None => 0
    }
    println("a.n = nested scrutinee Some(7), norm 7")
    println(a.n)
    mut i = 0
    xs[i] += match found(i) {
        Some(v) => v
        None => 500
    }
    println("xs[0] += none, norm 1503")
    println(xs[0])
}
