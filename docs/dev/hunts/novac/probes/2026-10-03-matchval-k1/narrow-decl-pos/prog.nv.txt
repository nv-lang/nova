module prog

// D489: a literal takes the type of its position when its value fits; a
// declared local type is such a position (`ro x T = e`), and the `match`
// value's arms take the type their neighbour value has. Each print is
// labelled with the norm.
fn found(k int) -> Option[int] {
    if k > 0 { Some(k) } else { None }
}

fn take8(x u8) -> int => x as int

fn main() {
    ro k = 3
    ro a u8 = 200 as u8
    println("a u8 = 200, (a + 55) as int, norm 255")
    println((a + 55) as int)
    ro b i16 = -300 as i16
    println("b i16 = -300, norm -300")
    println(b as int)
    ro o Option[int] = found(k)
    println("o Option[int] = found(3), norm 3")
    println(o ?? 0)
    ro f f64 = 3 as f64
    println("f f64 = 3, f / 2.0, norm 1.5")
    println(f / 2.0)
    ro w u8 = a
    println("w u8 = a, norm 200")
    println(w as int)
    println("take8(match: a or 7), norm 200")
    println(take8(match k {
        3 => a
        _ => 7 as u8
    }))
    println("take8(match Some(v) => a, None => 255), norm 255")
    println(take8(match found(0) {
        Some(v) => a
        None => 255 as u8
    }))
    ro e = match k {
        3 => a
        _ => 0 as u8
    }
    println("e = match a or 0, (e + 55) as int, norm 255")
    println((e + 55) as int)
    ro g u8 = match k {
        3 => a
        _ => w
    }
    println("g u8 = match a or w, norm 200")
    println(g as int)
}
