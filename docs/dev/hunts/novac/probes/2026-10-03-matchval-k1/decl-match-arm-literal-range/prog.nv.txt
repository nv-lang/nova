module prog

// A typed binding whose initializer is a `match` with LITERAL arms (task 5 of
// mono-k2: the declared type is held to the initializer's). D489: every arm
// is a position of the written type; a literal that does not fit is
// E_LIT_OUT_OF_RANGE ("no implicit narrowing, truncation or rounding that
// changes the WRITTEN number"). Here 300 and -1 stand in a `u8` arm and 40000
// in an `i16` arm. The norm: no program. Printed value if one is built: the
// truncated number (44, 255, -25536).
fn main() {
    ro k = 3
    ro c u8 = match k {
        3 => 300
        _ => 0
    }
    println(c as int)
    ro e u8 = match k {
        3 => -1
        _ => 0
    }
    println(e as int)
    ro d i16 = match k {
        3 => 40000
        _ => 0
    }
    println(d as int)
}
