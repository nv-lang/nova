module prog

// `x += match ..` whose taken arm WRITES `x` before yielding its value. Which
// `x` the compound reads (before or after the arm ran) is a measurement here,
// not a norm (no D-block found that fixes the order of a compound store's
// read against its right side); the two compilers are compared.
fn main() {
    ro k = 3
    mut x = 1
    x += match k {
        3 => {
            x = 100
            1
        }
        _ => 0
    }
    println(x)
    mut ys = [1, 2]
    ys[0] += match k {
        3 => {
            ys[0] = 50
            1
        }
        _ => 0
    }
    println(ys[0])
}
