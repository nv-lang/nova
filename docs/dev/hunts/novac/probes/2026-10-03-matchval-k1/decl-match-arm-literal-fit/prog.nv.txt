module prog

// The in-range twin of decl-match-arm-literal-range: literal arms of a
// `match` under a declared local type, every literal fitting. D489: each arm
// takes the written type. The norm: 250, then 255 (c + 5 computed in u8),
// then -1, then 32767.
fn found(k int) -> Option[int] {
    if k > 0 { Some(k) } else { None }
}

fn main() {
    ro k = 3
    ro c u8 = match k {
        3 => 250
        _ => 0
    }
    println(c as int)
    println((c + 5) as int)
    ro d i16 = match found(k) {
        Some(v) => -1
        None => 7
    }
    println(d as int)
    ro e i16 = match found(0) {
        Some(v) => 1
        None => 32767
    }
    println(e as int)
}
