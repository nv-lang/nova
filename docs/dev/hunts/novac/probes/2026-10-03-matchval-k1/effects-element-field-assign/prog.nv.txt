module prog

// Effects around a `match` value in the positions D484 does NOT name (its
// section 5: "a constructor and a record literal are not decided by this
// block"): an array element, a constructor field, the right of `+=` with an
// effectful index, a nested `match` as scrutinee. Measured against the
// oracle; the count of each `say` line is what matters first -- every
// expression is evaluated exactly once whatever the order.
fn say(n int) -> int {
    println(n)
    n
}

type Trio { a int, b int, c int }

fn main() {
    println("array element: each of 1 2 3 once")
    ro xs = [say(1), match say(2) {
        2 => 2
        _ => 0
    }, say(3)]
    println(xs[0] * 100 + xs[1] * 10 + xs[2])
    println("ctor field: each of 4 5 6 once")
    ro t = Trio { a: say(4), b: match say(5) {
        5 => 5
        _ => 0
    }, c: say(6) }
    println(t.a * 100 + t.b * 10 + t.c)
    println("+= with effectful index: each of 0 and 7 once, ys[0] 8")
    mut ys = [1, 2, 3]
    ys[say(0)] += match say(7) {
        7 => 7
        _ => 0
    }
    println(ys[0])
    println("nested scrutinee: 9 once, then 1")
    ro r = match match say(9) {
        9 => 1
        _ => 2
    } {
        1 => 1
        _ => 0
    }
    println(r)
}
