module p09_std_sum_fenced_unit.main

// A std payload-free sum, laid out by value in the shell, fenced by this
// single file: `[]PathStyle`. Unit mode -- no program root.

import std.fs.{PathStyle}

fn style_name(s PathStyle) -> str => match s {
    Posix => "posix"
    Windows => "windows"
}

fn main() {
    mut xs = []PathStyle.new()
    xs.push(PathStyle.Posix)
    xs.push(PathStyle.Windows)
    for s in xs {
        println(style_name(s))
    }
}
