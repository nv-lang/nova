module p05_value_record_field.bag

import ../pal.{Color}

/// How many times `c` is in a vector of three -- the fence.
export fn count(c Color) -> int {
    mut xs = []Color.new()
    xs.push(Color.Red)
    xs.push(Color.Green)
    xs.push(Color.Red)
    mut n = 0
    for x in xs {
        if x == c { n += 1 }
    }
    n
}
