// aggregate.nv — the example's real code (excerpt; real-cancel supervised(deadline:))fn fetch_guarded(source Source, budget_ms int, idx int) Net Time Emit -> TaskResult {    ro t0 Monotonic = Monotonic.now()    Emit.lane_started(source.id, idx)    ro outcome    = fetch_one(source, idx)          // a real Net round-trip    ro elapsed_ms = deadline_elapsed_ms(t0)    ro status = if elapsed_ms > budget_ms {        TaskStatus.Cancelled                        // too late — over budget    } else {        match outcome {            Ok(data) => TaskStatus.Done(data)            Err(e)   => TaskStatus.Failed(e.to_str())        }    }    // + Emit.lane_done / lane_failed / lane_cancelled — see the full file    TaskResult { id: source.id, status, elapsed_ms }}export fn aggregate(sources []Source, budget Duration) Net Time Emit -> Report {    ro t0 Monotonic = Monotonic.now()    ro budget_ms = budget.as_millis() as int    ro results = parallel for idx int in 0..sources.len() {   // fan-out        fetch_guarded(sources[idx], budget_ms, idx)    }    Emit.swarm_done(…)    Report.from(results, deadline_elapsed_ms(t0))}
// aggregate_test.nv — excerpt: we swap the WORLD, not the code (zero mocking frameworks)test "aggregate: done/failed/cancelled settle correctly on a mixed fan-out" {    with Net = mock_net(), Emit = null_emit() {   // ← the whole network world = one handler        ro report = aggregate(tiny_sources(), Duration.from_millis(120))        assert(report.done == 2)        assert(report.failed == 1)        assert(report.cancelled == 1)        assert(report.wall_ms < report.sequential_ms)    // speedup    }}// the same aggregate under with Net = real_net() — Live; the code does not change
